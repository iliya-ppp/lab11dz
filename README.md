# Домашнее задание к работе 11

## Условие задачи
Создание структуры «Авиарейс» (Вариант 18) для хранения данных 5 рейсов и поиска варианта с максимальной скоростью. Результат выводится в прямоугольном окне (10 строк на 80 символов) с использованием `sprintf()`. 

**Поля записи:** номер рейса, время вылета/прилета (часы, минуты), направление, марка самолета, расстояние (км).

## 1. Алгоритм и блок-схема
### Алгоритм
1. Инициализация локали и объявление структуры `Flight` / массива из 5 элементов.
2. Ввод данных с клавиатуры и перевод времени в минуты от начала суток.
3. Расчет длительности полета (с учетом перехода через полночь), вычисление скорости и поиск максимального значения.
4. Форматирование строк через `sprintf()` и вывод в графической рамке.

## 2. Реализация программы
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <string.h>
#include <locale.h>
#include <windows.h>

struct Flight {
    char num[20];
    int vilet;    
    int prilet;   
    char napr[50];
    char samolet[50];
    double dist;  
};
typedef struct Flight F;

int main() {
    setlocale(LC_ALL, "Russian");
    F f[5]; 

    for (int i = 0; i < 5; i++) {
        int v_h, v_m, p_h, p_m;
        printf("\nВведите рейс %d (Номер, часы вылета минуты вылета, часы прилета минуты прилета, направление, самолет, расстояние):\n", i + 1);
        scanf("%s %d %d %d %d %s %s %lf", f[i].num, &v_h, &v_m, &p_h, &p_m, f[i].napr, f[i].samolet, &f[i].dist);
        
        f[i].vilet = v_h * 60 + v_m;
        f[i].prilet = p_h * 60 + p_m;
    }

    int max_idx = 0;
    double max_speed = 0;
    for (int i = 0; i < 5; i++) {
        int t = f[i].prilet - f[i].vilet;
        if (t <= 0) t += 24 * 60; 
        
        double speed = f[i].dist / ((double)t / 60.0);
        if (speed > max_speed) {
            max_speed = speed;
            max_idx = i;
        }
    }

    char s1[100], s2[100];
    sprintf(s1, "рейс: %s | направление: %s", f[max_idx].num, f[max_idx].napr);
    sprintf(s2, "самолёт: %s | скорость: %.1f км/ч", f[max_idx].samolet, max_speed);

    printf("\n********************************************************************************\n"); 
    printf("*                                                                              *\n"); 
    printf("*                                                                              *\n"); 
    printf("*                                                                              *\n"); 
    printf("* %-76s *\n", s1);                                                                     
    printf("* %-76s *\n", s2);                                                                     
    printf("*                                                                              *\n"); 
    printf("*                                                                              *\n"); 
    printf("*                                                                              *\n"); 
    printf("********************************************************************************\n"); 

    return 0;
}
```
